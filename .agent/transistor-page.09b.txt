    if (ratio < .06) return `${name} is almost lost on the ${coin.label}: roughly one ${Math.round(1/ratio)}th of its diameter.`;
    if (ratio < .18) return `${name} spans only ${Math.round(ratio*100)}% of the ${coin.label}—several fit side by side.`;
    if (ratio < .42) return `${name} spans about ${ratio < .29 ? "one quarter" : "one third"} of the ${coin.label}.`;
    if (ratio < .72) return `${name} takes up a little over half the ${coin.label}'s diameter.`;
    if (ratio < .94) return `${name} nearly fills the ${coin.label}, with just a narrow rim to spare.`;
    if (ratio <= 1.08) return `${name} and the ${coin.label} are remarkably close in their largest dimension.`;
    if (ratio < 1.5) return `${name} overhangs the ${coin.label}: it is ${ratio.toFixed(1)} times the coin's diameter.`;
    return `${name} dwarfs the ${coin.label}, stretching ${ratio.toFixed(1)} times across it.`;
  }
