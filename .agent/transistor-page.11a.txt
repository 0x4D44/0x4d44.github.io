      const x = positions[i];
      const scale = ppm;
      const artY = baseY - pkg.h*scale/2 - (pkg.kind === "power" || pkg.kind === "to92" ? Math.min(pkg.h*.17*scale, 24) : 0);
      html += `<g transform="translate(${x} ${artY}) scale(${scale})">${packageMarkup(pkg, 0, 0, 1, true, `line-${pkg.id}-`)}</g>
        <text class="lineup-name" x="${x}" y="${baseY+22}">${pkg.short}</text>
        <text class="lineup-size" x="${x}" y="${baseY+35}">${fmt(spanOf(pkg))} mm</text>`;
    });
    svg.innerHTML = html;
  }

  function buildControls() {
    $("#packageRail").innerHTML = PACKAGES.map(p => `<button type="button" class="chip" data-id="${p.id}" aria-pressed="${p.id === state.packageId}">${p.short}</button>`).join("");
    $("#coinRail").innerHTML = COINS.map(c => `<button type="button" class="chip coin-chip" data-id="${c.id}" aria-pressed="${c.id === state.coinId}">${c.label} · ${c.diameter.toFixed(2)} mm</button>`).join("");
    $("#calibrationRange").value = state.calibration.toFixed(2);
    $("#calibrationOutput").textContent = `${state.calibration.toFixed(2)} px/mm`;
  }

  function setPackage(id) {
    state.packageId = id;
    state.dx = 0;
    state.dy = 0;
    renderStage();
  }
  function setCoin(id, scroll = false) {
    state.coinId = id;
    renderStage();
    if (scroll) $("#compare").scrollIntoView({ behavior: matchMedia("(prefers-reduced-motion: reduce)").matches ? "auto" : "smooth", block: "start" });
  }

  renderLineup();
  renderStage();
