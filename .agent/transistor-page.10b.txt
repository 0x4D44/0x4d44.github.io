  function simplePackageMarkup(pkg) {
    // Re-use the full art, but drop the stage-only group id in miniature views.
    return packageMarkup(pkg).replace(' id="package"', '').replace(/url\(#(?:blackPlastic|brushedMetal|darkMetal|leadMetal)\)/g, m => {
      if (m.includes("blackPlastic")) return "#11191d";
      if (m.includes("darkMetal")) return "#758187";
      if (m.includes("leadMetal")) return "#b8c2c5";
      return "#c9d1d3";
    }).replace(/filter="url\(#softShadow\)"/g, "");
  }
