      <text class="coin-text" y="${(-r*.04).toFixed(2)}" font-size="${textSize.toFixed(2)}">${coin.label}</text>
      <text class="coin-microtext" y="${(r*.47).toFixed(2)}" font-size="${Math.max(.7, d*.055).toFixed(2)}">UK · ${coin.diameter.toFixed(2)} mm</text>

  }

  function leadRects(width, bodyH, count = 3, leadLength = null) {
    const len = leadLength || Math.max(1.2, bodyH * .45);
    const leadW = Math.max(.18, width * .07);
    const gap = count === 1 ? 0 : width * .56 / (count - 1);
    let out = "";
    for (let i = 0; i < count; i++) {
      const x = count === 1 ? 0 : -width*.28 + gap*i;
      out += `<rect class="pkg-lead" x="${(x-leadW/2).toFixed(3)}" y="${(bodyH/2-.08).toFixed(3)}" width="${leadW.toFixed(3)}" height="${len.toFixed(3)}" rx="${(leadW*.22).toFixed(3)}"/>`;
    }
    return out;
  }
