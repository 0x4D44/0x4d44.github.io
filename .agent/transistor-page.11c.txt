    stage.setPointerCapture(event.pointerId);
    stage.classList.add("dragging");
  });
  stage.addEventListener("pointermove", event => {
    if (!dragging) return;
    const p = svgPoint(event);
    state.dx = p.x - grabOffset.x;
    state.dy = p.y - grabOffset.y;
    $("#packageLayer").innerHTML = packageMarkup(pkgById(state.packageId), state.dx, state.dy);
    $("#dimensionLayer").innerHTML = dimensionMarkup(pkgById(state.packageId), coinById(state.coinId), state.dx, state.dy);
  });
  function stopDrag(event) {
    if (!dragging) return;
    dragging = false;
    if (stage.hasPointerCapture(event.pointerId)) stage.releasePointerCapture(event.pointerId);
    stage.classList.remove("dragging");
  }
  stage.addEventListener("pointerup", stopDrag);
  stage.addEventListener("pointercancel", stopDrag);

  window.addEventListener("keydown", event => {
    if (event.key === "Escape") {
      state.dx = 0; state.dy = 0; renderStage();
    }
  });
})();
